```rust
struct TbcLongTermEconomics {
    // Fixed supply creates appreciation
    max_supply: u64,                    // 100,000,000 TBC - NEVER changes

    // Distribution complete by Year 10
    year_10_distribution: TbcDistribution {
        player_rewards: 40_000_000,     // 40% - Earned through play
        treasury: 15_000_000,           // 15% - Protocol reserve
        team: 15_000_000,               // 15% - Fully vested by Year 5
        ecosystem: 15_000_000,          // 15% - Grants, partnerships
        public_sale: 10_000_000,        // 10% - Initial and ongoing sales
        burned: 5_000_000,              // 5%  - Permanently destroyed
```

### 3.3 Validator Economics

```rust
struct ValidatorEconomics {
    // Staking requirements
    minimum_stake: u64,                 // 100,000 TBC ($500M at Y20 prices)
    optimal_stake: u64,                 // 500,000 TBC
    maximum_stake: u64,                 // 2,000,000 TBC (anti-concentration)

    // Reward structure
    fn calculate_rewards(&self, stake: u64, protocol_fees: u64) -> ValidatorRewards {
        // Validators share 50% of protocol fees
        let fee_pool = protocol_fees / 2;

        // Proportional to stake (with diminishing returns above optimal)
        let effective_stake = if stake > self.optimal_stake {
            self.optimal_stake + (stake - self.optimal_stake).sqrt()
        } else {
            stake
        };

        let stake_share = effective_stake as f64 / self.total_staked as f64;
        let fee_reward = (fee_pool as f64 * stake_share) as u64;

        // Additional rewards for uptime, performance
        let performance_bonus = self.calculate_performance_bonus();

        ValidatorRewards {
            fee_reward,
            performance_bonus,
            total: fee_reward + performance_bonus,
            apy: self.calculate_apy(stake, fee_reward + performance_bonus),
        }
    }
}
```
